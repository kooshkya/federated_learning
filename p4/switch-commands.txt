# --- Multicast group 1: all three ports ---
# Step 1: create multicast group 1
mc_mgrp_create 1
# Step 2: create a node with rid=0, port list = 1 2 3
mc_node_create 0 1 2 3
# Step 3: associate node handle 0 to group 1
mc_node_associate 1 0
